#!/usr/bin/env python # -*- coding: utf-8 -*- """ build_board.py ============== Reads the database, the template and the player sheets in Scouting\\files and writes the scouting board twice, with whatever is in the folder right now: files\\board.html, the copy you keep open locally, and github\\index.html, the copy you upload to GitHub Pages. Identical files — upload the second one and the published board matches what you see on your screen. Double-click it. Then open board.html in your browser and bookmark it: from then on, refreshing the board is one double-click and one F5. The same board.html works three ways with no changes: opened from disk, put on any static host, or sent to someone as a file. Everything it needs is inside it. It never reads Excel's computed cells, so it works on a sheet you have just generated and never opened. It takes the raw FotMob totals from the database, your competency scores from each sheet, and the model itself — benchmarks, competency weights, the data/scout split — from the template, then does the arithmetic here. Edit Benchmarks or Competency Map and the board follows, the same way the sheets do. Requires: openpyxl (python -m pip install openpyxl) """ import datetime import json import sys import traceback from pathlib import Path try: import openpyxl except ImportError: print("openpyxl is missing. Run: python -m pip install openpyxl") if len(sys.argv) == 1: input("\nPress Enter to close...") sys.exit(1) BASE = Path(__file__).resolve().parent.parent # Scouting\ FILES = BASE / "files" DB = FILES / "scouting_database_fotmob.xlsx" TPL = FILES / "player_sheet_template.xlsx" OUT = FILES / "board.html" # the copy you bookmark and open locally PUBLISH = BASE / "github" / "index.html" # the copy you upload to GitHub Pages ROLE_SHEETS = ["GK", "CB", "FB", "CM-DM", "AM-W", "ST"] COMP_FIRST, COMP_ROWS = 7, 8 # Competencies!B7:B14 WEIGHT_CELL = "H2" # Rating Breakdown — weight of the data side MONTHS = ["gennaio", "febbraio", "marzo", "aprile", "maggio", "giugno", "luglio", "agosto", "settembre", "ottobre", "novembre", "dicembre"] def stamp(): t = datetime.date.today() return f"{t.day} {MONTHS[t.month - 1]} {t.year}" def clamp(v): return max(0.0, min(100.0, v)) def r1(v): return round(v, 1) if isinstance(v, (int, float)) else None # ----------------------------------------------------------------- the model def read_model(tpl): """Benchmarks, Competency Map and the data/scout split, straight from the template.""" wb = openpyxl.load_workbook(tpl, data_only=True) for s in ("Benchmarks", "Competency Map", "Rating Breakdown"): if s not in wb.sheetnames: wb.close() raise SystemExit(f"{tpl.name} has no '{s}' sheet — is it the current template?") bench = {} for row in wb["Benchmarks"].iter_rows(min_row=3, values_only=True): role, metric, mtype, direction, mn, mx, weight = row[1:8] if not role or role not in ROLE_SHEETS: continue bench.setdefault(role, []).append( dict(n=metric, t=mtype, d=direction, mn=mn, mx=mx, w=weight)) comps = {} for row in wb["Competency Map"].iter_rows(min_row=3, values_only=True): prof, comp, weight = row[1], row[2], row[4] if not prof or not comp: continue comps.setdefault(prof, []).append(dict(n=comp, w=weight)) dw = wb["Rating Breakdown"][WEIGHT_CELL].value wb.close() if not isinstance(dw, (int, float)): dw = 45 return bench, comps, float(dw) # -------------------------------------------------------------- the database def read_database(db): wb = openpyxl.load_workbook(db, data_only=True) rows = [] for sheet in ROLE_SHEETS: if sheet not in wb.sheetnames: continue ws = wb[sheet] headers = [c.value for c in ws[2]] for r in ws.iter_rows(min_row=3, values_only=True): if not r[0]: continue rec = {h: v for h, v in zip(headers, r) if h} if "example row" in str(rec.get("Scout notes") or ""): continue rec["_sheet"] = sheet rows.append(rec) wb.close() return rows # ------------------------------------------------- the scout's own judgement def read_player_sheet(path): """Only cells you typed — never a computed one.""" wb = openpyxl.load_workbook(path, data_only=True) if "Profile" not in wb.sheetnames: wb.close() return None p = wb["Profile"] out = dict(name=p["B3"].value, cur=p["D13"].value, pot=p["D14"].value, scores=[]) if "Competencies" in wb.sheetnames: co = wb["Competencies"] for i in range(COMP_ROWS): out["scores"].append(co.cell(row=COMP_FIRST + i, column=2).value) wb.close() return out # ------------------------------------------------------------- the arithmetic def score_objective(rec, metrics): minutes = rec.get("Minutes") out, got, tot = [], 0.0, 0.0 for m in metrics: raw = rec.get(m["n"]) used = s = None if isinstance(raw, (int, float)): if m["t"] == "per90": if isinstance(minutes, (int, float)) and minutes > 0: used = raw / minutes * 90 else: used = raw if used is not None and m["mx"] != m["mn"]: span = m["mx"] - m["mn"] s = clamp((used - m["mn"]) / span * 100) if m["d"] == 1 \ else clamp((m["mx"] - used) / span * 100) got += s * m["w"] / 100.0 tot += m["w"] out.append(dict(n=m["n"], raw=raw, used=round(used, 2) if used is not None else None, mn=m["mn"], mx=m["mx"], s=round(s, 1) if s is not None else None, w=m["w"])) return (got / tot * 100 if tot else None), round(tot), out def score_subjective(scores, comps): out, got, tot = [], 0.0, 0.0 for i, c in enumerate(comps): v = scores[i] if i < len(scores) else None if not isinstance(v, (int, float)): v = None if v is not None: got += clamp(v / 10 * 100) * c["w"] / 100.0 tot += c["w"] out.append(dict(n=c["n"], v=v, w=c["w"])) return (got / tot * 100 if tot else None), round(tot), out def combine(obj, sub, data_weight): if obj is None and sub is None: return None if obj is None: return sub if sub is None: return obj return data_weight / 100 * obj + (100 - data_weight) / 100 * sub # -------------------------------------------------------------------- build def norm(s): return "".join(ch for ch in str(s or "").lower() if ch.isalnum()) def build(): here = Path(__file__).resolve().parent head = (here / "board_shell_head.html").read_text(encoding="utf-8") tail = (here / "board_shell_tail.html").read_text(encoding="utf-8") for f in (DB, TPL): if not f.exists(): raise SystemExit(f"Not found: {f}\nThis script expects to sit in Scouting\\tools " "with the workbooks in Scouting\\files.") bench, comp_map, data_weight = read_model(TPL) rows = read_database(DB) if not rows: raise SystemExit("The database has no player rows yet.") judgement = {} for path in FILES.glob("player_sheet_*.xlsx"): if path.name == "player_sheet_template.xlsx" or path.name.startswith("~$"): continue try: j = read_player_sheet(path) except Exception as e: print(f" ! {path.name} could not be read ({e})") continue if j and j["name"]: judgement[norm(j["name"])] = j players, notes = [], [] for rec in rows: name = rec.get("Name") role = rec["_sheet"] prof = rec.get("Comp profile") j = judgement.get(norm(name), dict(cur=None, pot=None, scores=[])) obj, objw, mets = score_objective(rec, bench.get(role, [])) if prof and prof in comp_map: sub, subw, comps = score_subjective(j["scores"], comp_map[prof]) else: sub, subw, comps = None, 0, [] if not prof: notes.append(f"{name}: no 'Comp profile' in the database, so no scout score") if norm(name) not in judgement: notes.append(f"{name}: no player sheet in the folder, so no competency scores") players.append(dict( name=name, club=rec.get("Club"), league=rec.get("League"), born=rec.get("Birth Year"), nat=rec.get("Nationality"), foot=rec.get("Foot"), h=rec.get("Height (cm)"), mv=rec.get("Market value"), pos=role, prof=prof, team=rec.get("Team style"), cur=j["cur"], pot=j["pot"], prio=rec.get("Priority"), status=rec.get("Status"), season=rec.get("Season"), mins=rec.get("Minutes"), obj=r1(obj), objw=objw, sub=r1(sub), subw=subw, fin=r1(combine(obj, sub, data_weight)), mets=mets, comps=comps)) players.sort(key=lambda r: (r["fin"] is None, -(r["fin"] or 0))) payload = json.dumps(players, ensure_ascii=False, separators=(",", ":"), default=str) head = head.replace("peso dati 45%", f"peso dati {data_weight:.0f}%") head = head.replace("peso occhio 55%", f"peso occhio {100 - data_weight:.0f}%") head = head.replace("· 5 giocatori", f"· {len(players)} giocatori") head = head.replace("§§DATE§§", stamp()) page = head + payload + tail # A complete document: the board is opened from disk and served by a static # host, so it needs its own charset and viewport — nothing wraps it there. css_end = page.index("") + len("") document = ( '\n\n\n' '\n' '\n' + page[:css_end] + '\n\n\n' + page[css_end:] + '\n\n\n') # Both copies are written every run, byte for byte identical, so the local # board and the one you upload can never drift apart. written = [] for target in (OUT, PUBLISH): try: target.parent.mkdir(parents=True, exist_ok=True) target.write_text(document, encoding="utf-8") written.append(target) except OSError as e: print(f" ! could not write {target}: {e}") print() for w in written: print(f"written: {w}") print() print(f"model: {data_weight:.0f}% data / {100 - data_weight:.0f}% scout\n") for r in players: fin = f"{r['fin']:.1f}" if r["fin"] is not None else " —" miss = sum(1 for m in r["mets"] if m["s"] is None) flag = f" {miss} metric{'s' if miss != 1 else ''} missing" if miss else "" print(f" {fin:>5} {str(r['name']):<24} {str(r['prof'] or '—'):<4}{flag}") for n in dict.fromkeys(notes): print(" ! " + n) print("\nLocal: refresh board.html in the browser.") print("GitHub: upload github\\index.html over the old one — the link does not change.") if __name__ == "__main__": double_clicked = len(sys.argv) == 1 try: build() except SystemExit as e: if e.code not in (0, None): print(f"\n{e}") except Exception: print("\nSomething unexpected went wrong:\n") traceback.print_exc() if double_clicked: print() try: input("Press Enter to close...") except EOFError: pass
